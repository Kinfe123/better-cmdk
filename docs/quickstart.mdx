# Quickstart

This guide takes you from a fresh install to a fully functional command palette with keyboard shortcut support. You'll need better-cmdk installed and the stylesheet imported — see [Installation](/docs/installation) if you haven't done that yet.

## Add the component

Create a new component file and paste in the following. It defines four actions, registers the `⌘K` / `Ctrl+K` keyboard shortcut, and renders `CommandMenu`:

```tsx
"use client";

import { useState, useEffect } from "react";
import { CalendarIcon, SearchIcon, UserIcon, SettingsIcon } from "lucide-react";
import { CommandMenu, type CommandAction } from "better-cmdk";

const actions: CommandAction[] = [
  {
    name: "calendar",
    label: "Calendar",
    description: "Open calendar view",
    icon: <CalendarIcon className="size-4" />,
    group: "Suggestions",
    execute: () => console.log("Calendar selected"),
  },
  {
    name: "search",
    label: "Search",
    description: "Open search view",
    icon: <SearchIcon className="size-4" />,
    group: "Suggestions",
    execute: () => console.log("Search selected"),
  },
  {
    name: "profile",
    label: "Profile",
    description: "Open profile settings",
    icon: <UserIcon className="size-4" />,
    group: "Settings",
    shortcut: "⌘P",
    execute: () => console.log("Profile selected"),
  },
  {
    name: "settings",
    label: "Settings",
    description: "Open settings page",
    icon: <SettingsIcon className="size-4" />,
    group: "Settings",
    shortcut: "⌘S",
    execute: () => console.log("Settings selected"),
  },
];

export function CommandPalette() {
  const [open, setOpen] = useState(false);

  useEffect(() => {
    const down = (e: KeyboardEvent) => {
      if (e.key === "k" && (e.metaKey || e.ctrlKey)) {
        e.preventDefault();
        setOpen((open) => !open);
      }
    };

    document.addEventListener("keydown", down);
    return () => document.removeEventListener("keydown", down);
  }, []);

  return (
    <CommandMenu
      open={open}
      onOpenChange={setOpen}
      actions={actions}
      actionsPlaceholder="Search or ask AI..."
    />
  );
}
```

## Mount it in your app

Import and render `CommandPalette` anywhere in your component tree — typically at the root layout level so the keyboard shortcut is always active:

```tsx
import { CommandPalette } from "@/components/command-palette";

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html>
      <body>
        {children}
        <CommandPalette />
      </body>
    </html>
  );
}
```

Press `⌘K` (or `Ctrl+K` on Windows/Linux) to open the palette. Type to search, arrow keys to navigate, and `Enter` to run an action.

> **Note:** `CommandMenu` includes a built-in developer-trial AI chat endpoint at no cost, rate-limited to 10 requests per 10 minutes. Use `⌘ Enter` inside the palette to switch to chat mode. For production, set your own `chatEndpoint`. See [AI Responses](/docs/features/ai-responses) for details.
